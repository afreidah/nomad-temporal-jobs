"class" "topbar-button-prev" "href" $href "icon" "chevron-left" "onwidths" $onwidths "onwidthm" $onwidthm "onwidthl" $onwidthl "hint" (printf "%s (%s)" $prevTitle ($startarrow | safeHTML)) )}} {{- end }} {{- end }}
