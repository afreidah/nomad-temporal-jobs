"class" "topbar-button-next" "href" $href "icon" "chevron-right" "onwidths" $onwidths "onwidthm" $onwidthm "onwidthl" $onwidthl "hint" (printf "%s (%s)" $nextTitle ($endarrow | safeHTML)) )}} {{- end }} {{- end }}
